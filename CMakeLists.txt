cmake_minimum_required(VERSION 3.24)
project(Paperdesk VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Paperdesk uses the native Windows desktop APIs.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(paperdesk_core src/model.cpp src/storage.cpp)
target_include_directories(paperdesk_core PUBLIC src)
target_compile_definitions(paperdesk_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(paperdesk_core PUBLIC shell32 ole32)
add_executable(Paperdesk WIN32 src/main.cpp src/render.cpp src/app.rc)
target_link_libraries(Paperdesk PRIVATE paperdesk_core gdiplus comctl32 user32 gdi32 advapi32 dwmapi)
if(MSVC)
  target_compile_options(paperdesk_core PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
  target_link_options(Paperdesk PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
endif()
enable_testing()
add_executable(paperdesk_tests tests/core_tests.cpp)
target_link_libraries(paperdesk_tests PRIVATE paperdesk_core)
add_test(NAME core COMMAND paperdesk_tests)
add_executable(paperdesk_render_tests tests/render_tests.cpp src/render.cpp)
target_link_libraries(paperdesk_render_tests PRIVATE paperdesk_core gdiplus user32 gdi32)
add_test(NAME rendering COMMAND paperdesk_render_tests)
